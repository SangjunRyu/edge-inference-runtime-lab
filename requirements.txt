# Replaces plain onnxruntime in the same environment:
#   pip uninstall -y onnxruntime

# The openvino version must match the ABI onnxruntime-openvino was built
# against. 2026.3.1 loads but the provider fails with Windows error 127
# (missing procedure) and ONNX Runtime silently falls back to CPU.
# onnxruntime-openvino==1.24.1
# openvino==2025.4.1
#
# onnxruntime-openvino does not bundle the OpenVINO DLLs; bench/backends.py
# calls openvino.utils.add_openvino_libs_to_path() before creating a session.
# Then: --backend openvino --ov-device CPU | GPU | NPU
